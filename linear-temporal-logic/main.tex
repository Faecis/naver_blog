    \begin{center}
        {\large\bf Notes on Linear Temporal Logic} \\
    \end{center}

    \textbf{Syntax}

    \begin{itemize}
        \item Linear Temporal Logic (LTL) Formula:
        $$
            \varphi \Coloneqq p \mid \neg\varphi \mid \varphi\lor\varphi \mid \varphi\land\varphi \mid \varphi\imp\varphi \mid \iscircle\varphi \mid \lozenge\varphi \mid \square\varphi \mid \varphi \until \varphi
        $$
    \end{itemize}
    
    \textbf{Informal Definition}

    \begin{itemize}
        \item An atomic preposition $p$ holds if $p$ is true at the current point of time.
        \item The propositional connectives $\neg$, $\lor$, $\land$, and $\imp$ have their usual meanings.
        \item Meaning of Temporal Connectives:
        \begin{itemize}
            \item $\iscircle \varphi$: $\varphi$ holds next (at the next point in time).
            \item $\lozenge \varphi$: $\varphi$ holds eventually (now or at some point in the future).
            \item $\square \varphi$: $\varphi$ holds always (now and at all future points in time).
            \item $\varphi \until \psi$: $\varphi$ holds until $\psi$ holds ($\psi$ holds now or at some point of future, and $\varphi$ holds continuously until then).
        \end{itemize}
    \end{itemize}

    \textbf{More Formal Definition}

    \begin{itemize}
        \item Let $S$ be a set of states and $L: S \rightarrow \mathcal{P}(\texttt{Atoms})$ be a labeling function associating to each state $s$ a set of all atoms that are true in that state.
        \item Let $\pi$ be an infinite sequence of states $s_0, s_1, s_2, \cdots$.
        We think of $L(s_i)$ as the set of all atoms true at point $i$ in time on the path $\pi$.
        \item For each $i$, we write $\pi^i$ for the $i$-th suffix of $\pi$, namely $s_i, s_{i+1}, s_{i+2}, \cdots$ (e.g., $\pi^1 = s_1, s_2, s_3, \cdots$ and $\pi^2 = s_2, s_3, s_4, \cdots$).
        \item For an LTL formula $\varphi_1$ we define $\pi \models_L \varphi$, read \say{$\pi$ satisfies $\varphi$ with respect to labeling $L$} or \say{$\varphi$ holds for $\pi$ with respect to labeling $L$}, by structural recurson on $\varphi$:
        \begin{itemize}
            \item $\pi \models_L p$ iff. $p \in L(s_0)$,
            \item $\pi \models_L \varphi \land \psi$ iff. $\pi \models_L \varphi$ and $\pi \models_L \psi$,
            \item $\pi \models_L \varphi \lor \psi$ iff. $\pi \models_L \varphi$ or $\pi \models_L \psi$,
            \item $\pi \models_L \varphi \imp \psi$ iff. $\pi \models_L \varphi$ implies $\pi \models_L \psi$,
            \item $\pi \models_L \iscircle\varphi$ iff. $\pi^1 \models_L \varphi$,
            \item $\pi \models_L \lozenge\varphi$ iff. $\exists{i} \geq 0$ such that $\pi^i \models_L \varphi$,
            \item $\pi \models_L \square\varphi$ iff. $\forall{i} \geq 0$ such that $\pi^i \models_L \varphi$, \&
            \item $\pi \models_L \varphi \until \psi$ iff. $\exists{i} \geq 0$ such that $\pi^i \models_L \psi$ and $\forall{j} \in \{0, 1, \cdots, i-1\}$, we have $\pi^j \models_L \psi$.
        \end{itemize}
        \item $\models$ is called satisfaction relation.
        It is a relation between formulas and infinite sequences of states in the presence of a state labeling with atom sets.
        When the labeling $L$ is fixed, we can write $\pi \models \varphi$.
    \end{itemize}

    \textbf{Labeled Transition System}

    \begin{itemize}
        \item A labeled transition system (LTS) is a triple $\mathcal{M} = (S, \rightarrow, L)$ where:
        \begin{itemize}
            \item $S$ is a finite set of states, 
            \item ${\rightarrow} \subseteq S \times S$ is a transition relation, \& 
            \item $L: S \rightarrow \mathcal{P}(\texttt{Atoms})$ a labeling function,
        \end{itemize}
        such that every state has an outward transition (i.e., $\forall{s_1} \in S$, $\exists{s_2} \in S$ such that $s_1 \rightarrow s_2$).
        \item A path $\pi$ in an LTS $\mathcal{M} = (S, \rightarrow, L)$ is an infinite sequence of states $s_0, s_1, s_2, \cdots$ such that $\forall{i} \geq 0, s_1 \rightarrow s_{i+1}$.
        Paths are written as $\pi = s_0 \rightarrow s_1 \rightarrow s_2 \rightarrow \cdots$.
        \item For a state $s \in S$, we define $\mathcal{M}, s \models \varphi$, read \say{$\mathcal{M}$ satisfies $\varphi$ in state $s$} or \say{$\varphi$ holds for $\mathcal{M}$ in state $s$}, to mean that $\pi \models_L \varphi$ for every path $\pi$ of $\mathcal{M}$ starting at state $s$.
    \end{itemize}

    \textbf{Formula Equivalence}

    \begin{itemize}
        \item Two formulas $\varphi$ and $\psi$ are equivalent, denoted $\varphi \equiv \psi$, if they are satisfied by the same state labelings and infinite sequences of states: 
        given any labeling $L: S \rightarrow \mathcal{P}(\texttt{Atoms})$ and any infinite sequences of states $\pi$, we have that $\pi \models_L \varphi$ iff. $\pi \models_L \psi$.
        \item If $\varphi \equiv \psi$, then $\varphi$ and $\psi$ will also be satisfied by the same LTS's in the same states:
        given any LTS $\mathcal{M} = (S, \rightarrow, L)$ and any $s \in S$, we have that $\mathcal{M}, s \models \varphi$ iff. $\mathcal{M}, s \models \psi$.
